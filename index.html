async function submitId() { const fileInput = document.getElementById('idFile'); const msgEl = document.getElementById('idMsg'); if (!fileInput.files || fileInput.files.length === 0) { msgEl.className = 'error'; msgEl.innerText = 'Please select a clear ID image before proceeding.'; return; // Stops execution completely if no file is selected } const formData = new FormData(); formData.append('username', activeUsername); formData.append('idImage', fileInput.files[0]); msgEl.className = ''; msgEl.innerText = 'Uploading ID... Please wait.'; try { const res = await fetch(`${API_URL}/api/submit-id`, { method: 'POST', body: formData }); const data = await res.json(); if (res.ok) { msgEl.className = 'success'; msgEl.innerText = data.message; // Lock the screen so they cannot bypass document.getElementById('idFile').disabled = true; } else { msgEl.className = 'error'; msgEl.innerText = data.error || 'Upload failed. Try again.'; } } catch (err) { msgEl.className = 'error'; msgEl.innerText = 'Network error during upload. Please try again.'; } }
